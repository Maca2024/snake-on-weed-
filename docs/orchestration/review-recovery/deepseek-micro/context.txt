    snake: Array.from({ length: 4 }, (_, index) => ({ x: headX - index, y: headY })),
    food: null,
    score: 0,
    combo: 0,
    status: "ready",
    ticks: 0,
    foodAge: 0,
    foods: 0,
    _rng: (Number(seed) >>> 0) || 1,
  };
  if (!placeFood(state)) state.status = "won";
  return state;
}

export function queueDirection(state, dir) {
  if (state.status !== "running" || !Object.hasOwn(VECTORS, dir) || state.queue.length >= 2) return false;
  const previous = state.queue.at(-1) ?? state.dir;
  if (dir === previous || dir === OPPOSITE[previous]) return false;
  state.queue.push(dir);
